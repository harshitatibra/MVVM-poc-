cmake_minimum_required(VERSION 3.16)
project(quadruped_sim)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(qt_robot_interfaces REQUIRED)
add_executable(simulator src/main.cpp)
target_compile_features(simulator PRIVATE cxx_std_17)
target_include_directories(simulator PRIVATE include)
target_link_libraries(simulator PRIVATE rclcpp::rclcpp ${qt_robot_interfaces_TARGETS})
install(TARGETS simulator DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_test REQUIRED)
  add_executable(robot_state_test test/robot_state_test.cpp)
  target_compile_features(robot_state_test PRIVATE cxx_std_17)
  target_include_directories(robot_state_test PRIVATE include)
  ament_add_test(robot_state_test COMMAND "$<TARGET_FILE:robot_state_test>")
endif()
ament_package()
